cmake_minimum_required(VERSION 3.16)
project(PokemonRPG LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt5 5.12 REQUIRED COMPONENTS Widgets)
set(GAME_SOURCES BagWidget.cpp BarrierEditor.cpp BattleScene.cpp BoxGenerator.cpp
    ExplorationScene.cpp MenuScene.cpp PlayerItem.cpp PokemonItem.cpp scenemanager.cpp)
add_library(game STATIC ${GAME_SOURCES})
target_link_libraries(game PUBLIC Qt5::Widgets)
target_include_directories(game PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(game PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(ENABLE_SANITIZERS "Enable address and undefined behavior checks" OFF)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(game PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(game PUBLIC -fsanitize=address,undefined)
endif()
add_executable(PokemonRPG main.cpp Dataset/Image/resources.qrc)
target_link_libraries(PokemonRPG PRIVATE game)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt5 5.12 REQUIRED COMPONENTS Test)
    add_executable(game_tests tests/game_tests.cpp Dataset/Image/resources.qrc)
    target_link_libraries(game_tests PRIVATE game Qt5::Test)
    add_test(NAME game_tests COMMAND game_tests)
    set_tests_properties(game_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
endif()
